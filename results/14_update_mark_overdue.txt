-- [수정] 기한이 지났는데 미반납인 대여 건을 OVERDUE로 일괄 업데이트 후 상태별 건수 확인

$ sqlite> UPDATE rental SET status = 'OVERDUE' WHERE status = 'RENTED' AND return_date IS NULL AND due_date < date('now');
(영향받은 행 수: 5)


$ sqlite> SELECT status, COUNT(*) AS cnt FROM rental GROUP BY status;
status   | cnt
---------+----
OVERDUE  | 5  
RENTED   | 2  
RETURNED | 25 

(3 rows)
