-- [집계] 카테고리별 평균 도서 정가 (AVG + GROUP BY)

$ sqlite> SELECT cat.name AS category_name, ROUND(AVG(b.price), 0) AS avg_price, COUNT(b.book_id) AS book_count FROM book b INNER JOIN category cat ON cat.category_id = b.category_id GROUP BY cat.category_id, cat.name ORDER BY avg_price DESC;
