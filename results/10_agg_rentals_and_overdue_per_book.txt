-- [집계] 도서별 총 대여횟수/연체 발생횟수, 대여 많은 순 (COUNT+SUM + GROUP BY)

$ sqlite> SELECT b.title, COUNT(r.rental_id) AS total_rentals, SUM(CASE WHEN r.return_date IS NULL AND r.due_date < date('now') THEN 1 ELSE 0 END) AS overdue_count FROM rental r INNER JOIN book b ON b.book_id = r.book_id GROUP BY b.book_id, b.title ORDER BY total_rentals DESC LIMIT 10;
